# this script automates using gnuplot to create an image of some data
# Nathan Moore 2026-09-13

# sending output to png file
set terminal pngcairo size 800,600 enhanced font 'Arial,12'
set output "09-14-plot.png"

# reading data in from csv file
set datafile separator ","   # Tell gnuplot CSV uses commas

# fit a straight line to the data.  Fitting to the linear x, ln-y axes
# which maps to a decaying exponential function
f(x) = m*x + b
fit f(x) "09-14_Nathans_bubble_rate_over_days.csv" using ($1):(log(1/$2)) via m,b

# plot formatting
set xlabel 'day of September (day)'
set ylabel 'ln(bubble frequency) ln (1/frame)'
set grid

# run a plot command of data and fit function
plot "09-14_Nathans_bubble_rate_over_days.csv" using ($1):(log(1/$2)), f(x)
